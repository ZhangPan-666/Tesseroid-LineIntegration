cmake_minimum_required(VERSION 3.18)
project(TesseroidLineIntegration VERSION 0.1.0 LANGUAGES CXX)

include(CTest)
option(TLI_BUILD_OPENMP "Build the OpenMP Work drivers" OFF)
option(TLI_BUILD_MPI "Build pure-MPI Work and shell drivers" OFF)
option(TLI_BUILD_RESEARCH "Build the long-running research executables" OFF)

add_library(tli INTERFACE)
add_library(Tesseroid::tli ALIAS tli)
target_compile_features(tli INTERFACE cxx_std_17)
target_include_directories(tli INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/C++Codes")
add_library(tli_mmsi INTERFACE)
add_library(Tesseroid::mmsi ALIAS tli_mmsi)
target_compile_features(tli_mmsi INTERFACE cxx_std_17)
target_include_directories(tli_mmsi INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/TLI_MMSI")

function(tli_program name source library)
  add_executable(${name} "${source}")
  target_link_libraries(${name} PRIVATE ${library})
  if(MSVC)
    target_compile_options(${name} PRIVATE /utf-8 /EHsc)
    target_compile_definitions(${name} PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
endfunction()
